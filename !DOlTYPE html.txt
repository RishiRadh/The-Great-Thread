// Order cart (saved in this browser)
ART_KEY = 'mc_cart';
let cart = [];
try { cart = JSON.parse(localStorage.getItem(CART_KEY)) || []; } catch (e) { cart = []; }

function saveCart() {
  try { localStorage.setItem(CART_KEY, JSON.stringify(cart)); } catch (e) { /* storage blocked */ }
  renderCart();
}

function outOf(v) {
  return v.stock !== undefined && v.stock <= 0;
}

function addToCart(p, k) {
  const v = p.sizes[k];
  if (outOf(v)) { toast('That size is sold out'); return; }
  const have = (cart.find(l => l.sku === v.sku) || { qty: 0 }).qty;
  if (v.stock !== undefined && have >= v.stock) { toast(`Only ${v.stock} left in this size`); return; }
  const line = cart.find(l => l.sku === v.sku);
  if (line) line.qty += 1;
  else cart.push({ sku: v.sku, name: p.name, size: v.label, colour: p.colour, price: p.price, qty: 1 });
  saveCart();
  toast(`${p.name} (${v.label}) added to your order`);
}

function renderCart() {
  ount = cart.reduce((a, l) => a + l.qty, 0);
  $('#badge').textContent = count;
  $('#badge').hidden = !count;
  $('#lines').innerHTML = cart.length ? cart.map((l, i) => `
    <div class="line">
      <div><strong>${esc(l.name)}</strong><small>${esc(l.size)} · ${esc(l.colour)} · ${money(l.price)}</small></div>
      <div class="qty">
        <button type="button" data-q="${i}" data-d="-1" aria-label="Less">−</button>
        <span>${l.qty}</span>
        <button type="button" data-q="${i}" data-d="1" aria-label="More">+</button>
      </div>
    </div>`).join('') : '<p class="muted">Your order is empty. Add pieces from the collection.</p>';
  $('#total').textContent = money(cart.reduce((a, l) => a + l.qty * l.price, 0));
  $('#btnWa').disabled = !cart.length;
  $('#btnOrder').disabled = !cart.length;
  $('#btnCopy').disabled = !cart.length;
}

$('#lines').addEventListener('click', (e) => {

  if (!b) return;
  const i = +b.dataset.q;
  cart[i].qty += +b.dataset.d;
  if (cart[i].qty < 1) cart.splice(i, 1);
  saveCart();
});

// Build the order and its code. The billing software reads the code that starts with MC1:
function buildOrder() {
  const name = $('#cName').value.trim();
  const phone = $('#cPhone').value.trim();
  if (!cart.length) { toast('Add at least one piece first'); return null; }
  if (!name || !phone) { toast('Please enter your name and phone number'); return null; }
  return {
    v: 1,
    id: 'WEB' + Date.now().toString(36).toUpperCase(),
    date: today(),
    customer: name,
    phone,
    address: $('#cAddr').value.trim(),
    payment: $('#cPay').value,
    note: $('#cNote').value.trim(),
    items: cart.map(l => ({ sku: l.sku, name: l.name, size: l.size, colour: l.colour, price: l.price, qty: l.qty })),
    total: cart.reduce((a, l) => a + l.qty * l.price, 0)
  };
}

const encodeOrder = (o) => 'MC1:' + btoa(unescape(encodeURIComponent(JSON.stringify(o))));

function orderText(o, code) {
  const lines = o.items.map(i => `- ${i.name} (${i.size}, ${i.colour}) x${i.qty} = ${money(i.price * i.qty)}`);
  return [
    `New order ${o.id} for ${SHOP.name}`,
    `Name: ${o.customer}`,
    `Phone: ${o.phone}`,
    o.address ? `Address: ${o.address}` : '',
    `Payment: ${o.payment}`,
    o.note ? `Note: ${o.note}` : '',
    'Items:',
    ...lines,
    `Total: ${money(o.total)}`,
    '',
    code
  ].filter(x => x !== '').join('\n');
}

// ---------- Live connection to the shop database ----------
let photoMap = new Map();
let currentLogo = '';
let pendingLogo = null;

async function rest(path, opts = {}) {
  const headers = { apikey: CLOUD.anonKey, Authorization: 'Bearer ' + CLOUD.anonKey, 'Content-Type': 'application/json' };
  if (opts.owner) headers['x-shop-key'] = sessionStorage.getItem('mc_owner_key') || '';
  if (opts.prefer) headers.Prefer = opts.prefer;
  const res = await fetch(`${CLOUD.url}/rest/v1/${path}`, { method: opts.method || 'GET', headers, body: opts.body === undefined ? undefined : JSON.stringify(opts.body) });
  if (!res.ok) throw new Error(String(res.status));
  const text = await res.text();
  return text ? JSON.parse(text) : null;
}

function applySettings(r) {
  if (!r) return;
  SHOP.name = r.name || SHOP.name;
  SHOP.tagline = r.tagline || '';
  SHOP.phone = r.phone || '';
  SHOP.whatsapp = String(r.whatsapp || '').replace(/\D/g, '') || SHOP.whatsapp;
  SHOP.address = r.address || '';
  SHOP.hours = r.hours || '';
  currentLogo = r.logo || '';
  document.title = SHOP.name;
  document.querySelectorAll('[data-shop]').forEach(el => { el.textContent = SHOP[el.dataset.shop] || ''; });
  document.querySelectorAll('[data-wa]').forEach(el => { el.href = waLink(`Hello ${SHOP.name}, I would like to know more about your collection.`); });
  document.querySelectorAll('[data-tel]').forEach(el => { el.href = 'tel:' + SHOP.phone.replace(/[^+\d]/g, ''); });
  $('#logoImg').src = currentLogo;
  $('#logoImg').hidden = !currentLogo;
}

// One card per name and colour, with each size as an option (sizes out of stock are greyed out)
function groupProducts(rows) {
  const groups = new Map();
  rows.forEach(r => {
    const key = `${r.name}|${r.colour || ''}`;
    if (!groups.has(key)) groups.set(key, { name: r.name, cat: r.category || 'Other', colour: r.colour || '', price: Number(r.price) || 0, img: '', sizes: [] });
    const g = groups.get(key);
    g.sizes.push({ label: r.size || 'Free size', sku: r.sku, stock: Number(r.stock) || 0 });
    if (!g.img && photoMap.get(r.sku)) g.img = photoMap.get(r.sku);
    g.price = Math.min(g.price, Number(r.price) || g.price);
  });
  return [...groups.values()].map((g, i) => ({ ...g, i }));
}

async function loadLive(withPhotos) {
  const [settings, rows, offers] = await Promise.all([
    rest('shop_settings?id=eq.1&select=*'),
    rest('products?select=*&order=name.asc'),
    rest('offers?select=*&order=until.asc')
  ]);
  if (withPhotos) photoMap = new Map((await rest('product_photos?select=sku,photo') || []).map(r => [r.sku, r.photo]));
  applySettings(settings && settings[0]);
  PRODUCTS = groupProducts(rows || []);
  buildPills();
  renderProducts();
  renderOffers((offers || []).map(o => ({ ...o, text: o.details })));
}

async function placeOrder() {
  const o = buildOrder();
  if (!o) return;
  try {
    await rest('web_orders', { method: 'POST', prefer: 'return=minimal', body: {
      id: o.id, order_date: o.date, customer: o.customer, phone: o.phone, address: o.address,
      payment: o.payment, note: o.note, items: o.items, total: o.total, status: 'new' } });
    cart = [];
    saveCart();
    closeCart();
    toast(`Order ${o.id} placed. The shop has received it.`);
  } catch (err) {
    toast('Could not place the order. Please try again or use WhatsApp.');
  }
}

function shrinkDataUrl(src, maxSide) {
  return new Promise((resolve) => {
    const img = new Image();
    img.onload = () => {
      const scale = Math.min(1, maxSide / Math.max(img.width, img.height));
     anvas = document.createElement('canvas');
      canvas.width = Math.max(1, Math.round(img.width * scale));
      canvas.height = Math.max(1, Math.round(img.height * scale));
      canvas.getContext('2d').drawImage(img, 0, 0, canvas.width, canvas.height);
      resolve(canvas.toDataURL('image/webp', 0.85));
    };
    img.onerror = () => resolve('');
    img.src = src;
  });
}

async function saveAdmin() {
  const key = $('#ownerKey').value.trim();
  if (!key) { $('#adStatus').textContent = 'Enter your shop key.'; return; }
  if (!LIVE) { $('#adStatus').textContent = 'Connect the database first (see the setup guide).'; return; }
  sessionStorage.setItem('mc_owner_key', key);
  const row = {
    id: 1,
    name: $('#adName').value.trim() || SHOP.name,
    tagline: $('#adTagline').value.trim(),
    phone: $('#adPhone').value.trim(),
    whatsapp: $('#adWhats').value.replace(/\D/g, ''),
    address: $('#adAddress').value.trim(),
    hours: $('#adHours').value.trim(),
    logo: pendingLogo !== null ? pendingLogo : currentLogo,
    updated_at: new Date().toISOString()
  };
  try {
    await rest('shop_settings?on_conflict=id', { method: 'POST', owner: true, prefer: 'resolution=merge-duplicates,return=minimal', body: row });
    pendingLogo = null;
    applySettings(row);
    $('#adStatus').textContent = 'Saved. The billing software will update within a few seconds.';
  } catch (err) {
    $('#adStatus').textContent = 'Could not save. Check the shop key and your internet connection.';
  }
}
